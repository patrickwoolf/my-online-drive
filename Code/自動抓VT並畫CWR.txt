    function parseAdvancedText(text) {
        const lines = text.trim().split('\n');
        if (lines.length < 2) return [];

        let headerRow = lines[0].trim().split(/\s+/);
        let vo2Idx = headerRow.findIndex(h => h.includes("V'O2") || h.toUpperCase() === "VO2");
        let wrIdx = headerRow.findIndex(h => h.toUpperCase() === "WR");
        let vevo2Idx = headerRow.findIndex(h => h.includes("V'E/V'O2") || h.includes("VE/VO2"));
        let vevco2Idx = headerRow.findIndex(h => h.includes("V'E/V'CO2") || h.includes("VE/VCO2"));
        let rerIdx = headerRow.findIndex(h => h.toUpperCase() === "RER");

        if (vo2Idx === -1) vo2Idx = 0;
        if (wrIdx === -1) wrIdx = 4;

        const results = [];
        for (let i = 1; i < lines.length; i++) {
            const line = lines[i].trim();
            if (!line) continue;

            const cells = line.split(/\s+/);
            if (isNaN(parseFloat(cells[vo2Idx])) || isNaN(parseFloat(cells[wrIdx]))) continue;

            results.push({
                vo2: parseFloat(cells[vo2Idx]),
                wr: parseFloat(cells[wrIdx]),
                vevo2: vevo2Idx !== -1 ? parseFloat(cells[vevo2Idx]) : null,
                vevco2: vevco2Idx !== -1 ? parseFloat(cells[vevco2Idx]) : null,
                rer: rerIdx !== -1 ? parseFloat(cells[rerIdx]) : null
            });
        }
        return results;
    }

    function autoDetectThresholds(data) {
        let detectedVT1 = 80;
        let detectedVT2 = 130;

        let validVEVO2 = data.filter(d => d.vevo2 !== null && d.wr > 20 && d.wr < 180);
        if (validVEVO2.length > 0) {
            let minObj = validVEVO2.reduce((min, p) => p.vevo2 < min.vevo2 ? p : min, validVEVO2[0]);
            detectedVT1 = minObj.wr;
        }

        let validVT2Data = data.filter(d => d.wr > detectedVT1 && d.wr < 260);
        if (validVT2Data.length > 0) {
            let rerPoint = validVT2Data.find(d => d.rer >= 1.0);
            if (rerPoint) {
                detectedVT2 = rerPoint.wr;
            } else {
                let minVEVCO2 = validVT2Data.reduce((min, p) => p.vevco2 < min.vevco2 ? p : min, validVT2Data[0]);
                detectedVT2 = minVEVCO2.wr;
            }
        }

        return { vt1: detectedVT1, vt2: detectedVT2 };
    }

    function processCWRData(cwrData) {
        const groups = {};
        cwrData.forEach(item => {
            if (!groups[item.wr]) groups[item.wr] = [];
            groups[item.wr].push(item.vo2);
        });

        const averagedPoints = [];
        let logMsg = "【CWR 重複 4 次以上尾端數據統計】:\n";

        for (const wr in groups) {
            const vo2List = groups[wr];
            if (vo2List.length >= 4) {
                const lastFour = vo2List.slice(-4);
                const sum = lastFour.reduce((acc, val) => acc + val, 0);
                const avgVo2 = sum / 4;
                averagedPoints.push({ x: parseFloat(wr), y: parseFloat(avgVo2.toFixed(3)) });
                logMsg += `• 功率 ${wr}W (出現 ${vo2List.length} 筆) ➔ 最後 4 筆平均 VO₂: ${avgVo2.toFixed(3)} L/min\n`;
            }
        }
        return { points: averagedPoints.sort((a, b) => a.x - b.x), log: logMsg };
    }

    function processDataAndRenderChart(isFirstAutoDetect = false) {
        const testText = document.getElementById('test-raw-input').value;
        const cwrText = document.getElementById('test-cwr-input').value;

        globalTestData = parseAdvancedText(testText);
        const parsedCwr = parseAdvancedText(cwrText);
        const cwrResult = processCWRData(parsedCwr);
        globalCwrPoints = cwrResult.points;

        let vt1 = 80;
        let vt2 = 130;

        if (isFirstAutoDetect && globalTestData.length > 0) {
            const autoThresh = autoDetectThresholds(globalTestData);
            vt1 = autoThresh.vt1;
            vt2 = autoThresh.vt2;

            const wrValues = globalTestData.map(d => d.wr);
            const maxWr = Math.max(...wrValues, 240);
            
            document.getElementById('vt1-slider').max = maxWr;
            document.getElementById('vt2-slider').max = maxWr;
            document.getElementById('vt1-slider').value = vt1;
            document.getElementById('vt2-slider').value = vt2;
        } else {
            vt1 = parseInt(document.getElementById('vt1-slider').value);
            vt2 = parseInt(document.getElementById('vt2-slider').value);
        }

        document.getElementById('vt1-val-display').innerText = vt1 + " W";
        document.getElementById('vt2-val-display').innerText = vt2 + " W";
        document.getElementById('debug-log').innerText = `【系統通知】生理閾值自動判定完成。\n${cwrResult.log}`;

        renderChart(vt1, vt2);
    }
